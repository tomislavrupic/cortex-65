cmake_minimum_required(VERSION 3.25)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS")
set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Architecture")
project(Cortex65 VERSION 0.3.0 LANGUAGES C CXX)
enable_testing()
add_executable(EngineTests Tests/EngineTests.cpp)
target_include_directories(EngineTests PRIVATE Source)
add_test(NAME EngineTests COMMAND EngineTests)
add_executable(VoicePoolTests Tests/VoicePoolTests.cpp)
target_include_directories(VoicePoolTests PRIVATE Source)
add_test(NAME VoicePoolTests COMMAND VoicePoolTests)
add_executable(BinauralTests Tests/BinauralTests.cpp)
target_include_directories(BinauralTests PRIVATE Source)
add_test(NAME BinauralTests COMMAND BinauralTests)
set(JUCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/JUCE" CACHE PATH "JUCE source")
if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt")
  message(FATAL_ERROR "JUCE was not found. Initialize third_party/JUCE or configure -DJUCE_PATH=/path/to/JUCE.")
endif()
add_subdirectory("${JUCE_PATH}" JUCE)
juce_add_binary_data(CortexAnatomy HEADER_NAME AnatomyData.h NAMESPACE CortexAnatomy SOURCES Assets/brain_positions.bin Assets/brain_normals.bin)
juce_add_plugin(Cortex65 COMPANY_NAME "Pixel Records" BUNDLE_ID com.pixelrecords.cortex65 PLUGIN_MANUFACTURER_CODE Pxrc PLUGIN_CODE Cx65 PRODUCT_NAME "CORTEX 65" FORMATS AU Standalone IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE AU_MAIN_TYPE kAudioUnitType_MusicDevice)
target_sources(Cortex65 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(Cortex65 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Cortex65 PUBLIC juce::juce_audio_utils PRIVATE CortexAnatomy juce::juce_recommended_config_flags)
juce_add_console_app(ProcessorTests PRODUCT_NAME "Cortex Processor Tests")
target_sources(ProcessorTests PRIVATE Tests/ProcessorTests.cpp)
target_link_libraries(ProcessorTests PRIVATE Cortex65 CortexAnatomy)
add_test(NAME ProcessorTests COMMAND ProcessorTests)
add_test(NAME BinauralRenderTests COMMAND ProcessorTests --binaural-render)
add_test(NAME UserPresetTests COMMAND ProcessorTests --user-presets)
add_test(NAME HeadOrbitTests COMMAND ProcessorTests --head-orbit)
set_tests_properties(UserPresetTests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(AUWrapperTests Tests/AUWrapperTests.cpp)
target_link_libraries(AUWrapperTests PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
add_test(NAME AUWrapperTests COMMAND AUWrapperTests "$<TARGET_BUNDLE_DIR:Cortex65_AU>")
set_tests_properties(AUWrapperTests PROPERTIES SKIP_RETURN_CODE 77)
foreach(t Cortex65_AU Cortex65_Standalone)
add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:${t}>/Contents/Resources/Anatomy" COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Assets/ATTRIBUTION.md" "${CMAKE_CURRENT_SOURCE_DIR}/Assets/provenance.json" "$<TARGET_BUNDLE_DIR:${t}>/Contents/Resources/Anatomy" COMMAND /usr/bin/codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:${t}>" VERBATIM)
endforeach()
